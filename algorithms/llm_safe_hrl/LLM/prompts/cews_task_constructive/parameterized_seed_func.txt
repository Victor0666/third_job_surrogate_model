def get_task_priority_v2(
    min_exec_time,
    min_comm_time,
    min_incremental_energy,
    slack,
    upward_rank,
    remaining_work,
    ready_wait_time,
    uncertainty
):
    """Parameterized structural example; smaller scores have higher priority."""
    epsilon = PARAMS["epsilon"]

    def normalize(values):
        values = np.asarray(values, dtype=float)
        return values / (np.mean(np.abs(values)) + epsilon)

    duration = normalize(min_exec_time + min_comm_time)
    slack_pressure = normalize(slack)
    uncertainty_gate = normalize(uncertainty) * np.maximum(-slack_pressure, 0)
    score = (
        PARAMS["slack_weight"] * slack_pressure
        + PARAMS["energy_weight"] * normalize(min_incremental_energy)
        + PARAMS["duration_weight"] * duration
        - PARAMS["rank_weight"] * normalize(upward_rank)
        - PARAMS["work_weight"] * normalize(remaining_work)
        - PARAMS["wait_weight"] * normalize(ready_wait_time)
        - PARAMS["uncertainty_weight"] * uncertainty_gate
    )
    return np.nan_to_num(score)
